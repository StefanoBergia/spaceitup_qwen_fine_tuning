# d2 Endings ("proceed behind X") Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Routes whose end is out of sight are described up to where they disappear and end with "…, then proceed behind X" (or with their hidden turn), visible "well past" stops re-anchor to a nearby object, the d1 texts get their ending rewritten instead of recomposed, and eval / steering / audits understand the new endings.

CPU stages then rebuild v4 and stop at the G1e audit.

**Tech Stack:** Python 3.12, numpy, Pillow, pytest; `uv run` for everything.

**Spec:** `docs/superpowers/specs/2026-10-13-turn-events-d2-design.md`, Addendum T2 (and the d2 spec it amends).

# --- endings (d2 Addendum T2, 2026-10-14): a route that ends out of sight "proceeds behind" what hides it -------
HIDDEN_BLIP_M = 0.3      # a visible run this short between two hidden runs is flicker along an occluder's edge
FPV_STEP_M = 0.05        # fpv_paths.json points are the route resampled this far apart, counted back from its end


def hidden_tail_m(flags, step=ROUTE_STEP_M):
    """Length of a route's final hidden stretch, counted back from its last point, or None when that point is in view.
    `flags` holds one value per point, `step` metres apart (True = hidden). A visible run shorter than HIDDEN_BLIP_M
    between two hidden runs counts as hidden; a visible run at the very start never does."""
    f = [bool(x) for x in flags]
    if not f or not f[-1]:
        return None
    k = len(f) - 1
    while True:
        while k > 0 and f[k - 1]:
            k -= 1                       # k: first point of the current hidden run
        if k == 0:
            break
        i = k - 1
        while i > 0 and not f[i - 1]:
            i -= 1                       # i: first point of the visible run before it
        if i == 0 or (k - i) * step >= HIDDEN_BLIP_M:
            break                        # a visible start, or a real visible stretch
        k = i                            # a blip: absorb it and keep walking back
    return (len(f) - 1 - k) * step


def fpv_hidden_flags(sample_dir, candidate_index):
    """The dataset's own hidden flag per fpv point of one candidate (fpv_paths.json runs), or None without the file."""
    p = Path(sample_dir) / "fpv_paths.json"
    if not p.exists():
        return None
    runs = json.loads(p.read_text())["candidates"][candidate_index]["runs"]
    return [bool(r["hidden"]) for r in runs for _ in r["uv"]]
```

BEHIND_READ_M = (0.15, 0.3, 0.5, 0.8)   # where past the disappearance the blob hiding the route is read


def behind_anchor(frame, route_xz, disappear_arc_m):
    """The nameable blob a route goes out of sight behind, or None. The first blob a camera ray hits on the way to the
    route BEHIND_READ_M past `disappear_arc_m` decides; if it is not nameable, or no read point is hidden on the map,
    the blob hiding the route's end is tried. Nameable as a turn anchor is (_anchor_outline near the ray's hit)."""
    rg = route_geometry(route_xz)
    pts, arc = rg["pts"], rg["arc"]
    for ds in BEHIND_READ_M:
        s = disappear_arc_m + ds
        if s > arc[-1] + 1e-9:
            break
        occ, hit = _first_hit(frame, pts[min(int(np.searchsorted(arc, s)), len(pts) - 1)])
        if occ is not None:
            got = _anchor_outline(frame, occ, hit)
            if got:
                return {"blob": occ, "vis_xz": got[0], "outline": got[1]}
            break
    occ, hit = _first_hit(frame, pts[-1])
    if occ is not None:
        got = _anchor_outline(frame, occ, hit)
        if got:
            return {"blob": occ, "vis_xz": got[0], "outline": got[1]}
    return None

# the first waypoint of its final hidden run ("first_hidden") or the midpoint of that and the last visible one
# ("midpoint"). The CPU ceiling stage (prepare_habitat_instr.py endings-check) reports both; keep the one >= 0.95.
BEHIND_POINT = "first_hidden"


def point_polygon_distance(p, polygon):
    """Image distance from p to a polygon (0 inside); inf for a polygon with fewer than 3 vertices."""
    P = np.asarray(polygon if polygon is not None else [], float).reshape(-1, 2)
    if len(P) < 3:
        return float("inf")
    x, y = float(p[0]), float(p[1])
    inside = False
    for (x1, y1), (x2, y2) in zip(P, np.roll(P, -1, axis=0)):
        if (y1 > y) != (y2 > y) and x < (x2 - x1) * (y - y1) / (y2 - y1) + x1:
            inside = not inside
    if inside:
        return 0.0
    ab = np.roll(P, -1, axis=0) - P
    ap = np.array([x, y]) - P
    t = np.clip((ap * ab).sum(1) / np.maximum((ab * ab).sum(1), 1e-12), 0.0, 1.0)
    return float(np.min(np.linalg.norm(ap - t[:, None] * ab, axis=1)))


def behind_followed(path, outline, tol=STOP_TOL_UV, point=None):
    """1 when an image path ([x, y, v] waypoints) goes out of sight behind an object: its last waypoint is hidden
    (v = 0) and the BEHIND_POINT of its final hidden run lies inside `outline` dilated by tol. Where the hidden tail
    ends is not judged. Malformed paths or outlines give 0."""
    point = point or BEHIND_POINT
    try:
        if not path or len(path[-1]) < 3 or int(path[-1][2]) != 0:
            return 0
        k = len(path) - 1
        while k > 0 and len(path[k - 1]) >= 3 and int(path[k - 1][2]) == 0:
            k -= 1
        q = np.asarray(path[k][:2], float)
        if point == "midpoint" and k > 0:
            q = (q + np.asarray(path[k - 1][:2], float)) / 2
        return int(point_polygon_distance(q, outline) <= tol)
    except (TypeError, ValueError, IndexError):
        return 0

    """An answer path: solid where both ends of a segment are in view, dashed where it is out of sight."""
    W, H = img.size

    """G3, the label audit before the full job (d2 T2): passed d2 rows, ~rewritten_share of them d1 texts with a
    rewritten ending and the rest composed with the current prompt, half of those with a turn (turn rows fill what
    the no-turn pool cannot); reused d1 rows and rows of older prompts are left out."""
    rng = random.Random(seed)
    rows = [r for r in ok if r["id"] in events_by_id]
    rewritten = [r for r in rows if r.get("rewritten_from")]

           and not r.get("reused_from")]

    picked = rng.sample(rewritten, k_rw) + rng.sample(has, k) + rng.sample(no, min(m - k, len(no)))

        picked, pop = sample_endings(jsonl(args.events), desc, args.seed)
        items = [endings_item(by_id[r["id"]], r, objects, g,
                              text=desc[r["id"]]["description"] if g in ("behind", "visible") else None)
                 for g, r in picked]
        payload = {"items": items, "population": pop, "seed": args.seed, "mode": "endings",
                   "verdicts": VERDICTS["endings"], "source": str(args.events)}
        html = TEMPLATE.read_text().replace("/*__DATA__*/null", json.dumps(payload))
        html = html.replace("<title>Instruction Label Audit</title>", "<title>Endings Audit</title>", 1)
        out = args.out or REPO_ROOT / "outputs" / "endings_audit.html"
        out.parent.mkdir(parents=True, exist_ok=True)
        out.write_text(html)
        print(f"{len(items)} items (population {pop}) -> {out} ({out.stat().st_size / 1e6:.1f} MB)")
        return
```
